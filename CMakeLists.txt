cmake_minimum_required(VERSION 3.8)
project(rgbd_image_queue)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(message_filters REQUIRED)
find_package(image_transport REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/GetImage.srv"
  DEPENDENCIES sensor_msgs builtin_interfaces)
rosidl_get_typesupport_target(cpp_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_executable(rgbd_image_queue_node src/queue_node.cpp)
target_compile_features(rgbd_image_queue_node PRIVATE cxx_std_17)
target_compile_options(rgbd_image_queue_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(rgbd_image_queue_node PRIVATE include)
ament_target_dependencies(rgbd_image_queue_node rclcpp rcl_interfaces sensor_msgs message_filters image_transport)
target_link_libraries(rgbd_image_queue_node ${cpp_typesupport})
install(TARGETS rgbd_image_queue_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/rgbd_queue_client scripts/person_goal.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_pair_queue test/test_pair_queue.cpp)
  target_compile_features(test_pair_queue PRIVATE cxx_std_17)
  target_include_directories(test_pair_queue PRIVATE include)
  ament_target_dependencies(test_pair_queue sensor_msgs)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME rgbd_queue_ros_integration
    COMMAND ${Python3_EXECUTABLE} -s ${CMAKE_CURRENT_SOURCE_DIR}/test/test_ros_queue.py
      --binary $<TARGET_FILE:rgbd_image_queue_node>)
  set_tests_properties(rgbd_queue_ros_integration PROPERTIES TIMEOUT 70
    ENVIRONMENT "PYTHONNOUSERSITE=1;ROS_LOCALHOST_ONLY=1;ROS_DOMAIN_ID=229")
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
